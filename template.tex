% ---------------------------------------------------------------------------
%  A ruled monthly diary
%
%  Compiled by the `diary` command with Tectonic (XeTeX).
%  The body is generated from entries.md; everything outside the entries
%  block is safe to edit as a normal LaTeX template.
%
%  Page grid: A5, 22 lines per page, 7.2 mm between baselines.
%  The background rules are positioned to sit just below the text baselines,
%  so every line of writing rests on a rule.
% ---------------------------------------------------------------------------

\documentclass[11pt]{article}

% --- Page ------------------------------------------------------------------
\usepackage[
  a5paper,
  top=24mm, bottom=24mm, left=20mm, right=20mm,
  headheight=8mm, headsep=6mm, footskip=10mm
]{geometry}

% --- Type ------------------------------------------------------------------
% EB Garamond ships with the TeX Live bundle downloaded by Tectonic.
\usepackage{fontspec}
\setmainfont{EBGaramond}[
  Extension=.otf,
  UprightFont=*-Regular,
  ItalicFont=*-Italic,
  BoldFont=*-Bold,
  BoldItalicFont=*-BoldItalic,
  Numbers=OldStyle,
  Ligatures=TeX,
]

% A second family for the letterspaced small caps used in headings.
\newfontfamily\diarydisplay{EBGaramond}[
  Extension=.otf,
  UprightFont=*-Regular,
  ItalicFont=*-Italic,
  BoldFont=*-Bold,
  BoldItalicFont=*-BoldItalic,
  Numbers=OldStyle,
  LetterSpace=10,
]

\usepackage{microtype}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{fancyhdr}
\usepackage{needspace}
\usepackage[hidelinks]{hyperref}

% --- Palette ---------------------------------------------------------------
% Change these five colours to restyle the whole diary.
\definecolor{paper}{HTML}{FBF8F0}   % warm cream page
\definecolor{ink}{HTML}{2B2721}     % soft black text
\definecolor{rule}{HTML}{D8CCB8}    % the ruled lines
\definecolor{accent}{HTML}{9A5B4C}  % terracotta dates and ornaments
\definecolor{muted}{HTML}{8B8072}   % secondary text

% --- Page grid -------------------------------------------------------------
\setlength{\parindent}{1.5em}
\setlength{\parskip}{0pt}
\topskip=6mm
\clubpenalty=10000
\widowpenalty=10000
\emergencystretch=1.5em
\raggedbottom

% Header: a small editorial ornament, the month, and the year.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[C]{%
  \color{rule}\rule[0.31em]{10mm}{0.25pt}%
  \hspace{2.6mm}%
  {\diarydisplay\fontsize{8pt}{8mm}\selectfont
    \color{accent}\textsc{@@MONTH@@}\hspace{1.1mm}%
    {\color{muted}\textperiodcentered}\hspace{1.1mm}%
    \color{muted}@@YEAR@@}%
  \hspace{2.6mm}\color{rule}\rule[0.31em]{10mm}{0.25pt}%
}
\fancyfoot[C]{%
  \color{rule}\rule[0.31em]{8mm}{0.25pt}%
  \hspace{2.6mm}%
  {\diarydisplay\fontsize{8.5pt}{8mm}\selectfont\color{accent}\thepage}%
  \hspace{2.6mm}\color{rule}\rule[0.31em]{8mm}{0.25pt}%
}

% --- Ruled paper -----------------------------------------------------------
% 22 rules, starting 32.8 mm from the top and repeating every 7.2 mm.
% The first text baseline is at 30 mm (24 mm margin + 6 mm topskip).
\AddToHook{shipout/background}{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[paper] (current page.south west) rectangle (current page.north east);
    \foreach \i in {0,...,21} {
      \pgfmathsetmacro{\yy}{32.8+\i*7.2}
      \draw[rule, line width=0.22pt]
        ([xshift=18mm,yshift=-\yy mm]current page.north west) --
        ([xshift=130mm,yshift=-\yy mm]current page.north west);
    }
  \end{tikzpicture}%
}

% A small local time in the left margin at the start of an entry.
\newcommand{\entrytime}[1]{%
  \noindent
  \llap{{\fontsize{7.8pt}{7.2mm}\selectfont\itshape\color{muted}#1}\hspace{2.5mm}}%
  \ignorespaces
}

% One blank ruled line between separate entries on the same day.
\newcommand{\diaryentrygap}{%
  \par\vspace{\baselineskip}%
}

% --- Entries ---------------------------------------------------------------
% A date line in letterspaced small caps, with a PDF bookmark for navigation.
\newcounter{diaryentry}
\newcommand{\diarydate}[1]{%
  \stepcounter{diaryentry}%
  \pdfbookmark[1]{#1}{diary.\arabic{diaryentry}}%
  \noindent
  {\diarydisplay\fontsize{8.6pt}{7.2mm}\selectfont
    \color{accent}\textsc{#1}}%
  \par\nobreak
}

% One blank ruled line before a new date, with a guard against orphaned dates.
\newcommand{\diarygap}{%
  \par\needspace{3\baselineskip}\vspace{\baselineskip}%
}

\hypersetup{
  pdftitle={Diary - @@MONTH@@ @@YEAR@@},
  pdfsubject={A ruled monthly diary},
  pdfdisplaydoctitle=true,
}

\begin{document}
\fontsize{11pt}{7.2mm}\selectfont
\color{ink}

@@ENTRIES@@

\end{document}
